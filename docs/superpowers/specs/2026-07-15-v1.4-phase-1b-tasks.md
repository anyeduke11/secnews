  ```python
  def count_orphan_items(self) -> int:
      """无 concepts 关联的 items 数量。"""
      # concepts 为 '[]' 或 NULL
      conn = get_connection()
      row = conn.execute(
          "SELECT COUNT(*) FROM knowledge_items WHERE concepts IS NULL OR concepts = '[]'"
      ).fetchone()
      return row[0] if row else 0

  def count_stale_concepts(self, days: int = 30) -> int:
      """超过 N 天未更新的 concepts 数量。"""
      conn = get_connection()
      row = conn.execute(
          "SELECT COUNT(*) FROM knowledge_concepts WHERE updated_at < datetime('now', ?)",
          (f'-{days} days',),
      ).fetchone()
      return row[0] if row else 0

  def domain_coverage(self) -> list[dict]:
      """按 domain 分组统计覆盖度。"""
      conn = get_connection()
      rows = conn.execute("""
          SELECT
              COALESCE(domain, 'unknown') as domain,
              COUNT(*) as total,
              SUM(CASE WHEN compiled = 1 THEN 1 ELSE 0 END) as compiled
          FROM knowledge_items
          GROUP BY COALESCE(domain, 'unknown')
      """).fetchall()
      return [
          {"domain": r[0], "total": r[1], "compiled": r[2],
           "coverage": r[2]/r[1] if r[1] > 0 else 0}
          for r in rows
      ]
  ```

  ```python
  DEFAULT_SOUL = """---
  updated_at: "{timestamp}"

  # SOUL.md — 角色画像

  - 偏好难度: 待补充

  # SOUL 重新生成任务

  请扫描 knowledge/items/ 和 knowledge/concepts/，重新生成 SOUL.md 角色画像。
  ```

  ```python
  """Graph builder — construct knowledge graph from concepts + items."""
  import json
  from datetime import datetime, timezone, timedelta
  from typing import Optional
  from backend.repository.db import get_connection
  from backend.repository.knowledge_repo import knowledge_repo

  CACHE_TTL_MINUTES = 5

  def build_graph(domain: Optional[str] = None) -> dict:
      # 检查缓存
      cached = _get_cached_graph()
      if cached and _cache_fresh(cached, CACHE_TTL_MINUTES):
          graph = json.loads(cached["graph_data"])
          if domain:
              graph = _filter_by_domain(graph, domain)
          return graph

      # 构建节点
      concepts = knowledge_repo.list_concepts(domain=domain)
      nodes = [
          {"id": c.slug, "label": c.title, "domain": c.domain,
           "count": len(c.source_items), "wiki": "hotspot"}
          for c in concepts
      ]

      # 构建边（concept 共现）
      edges = _build_edges(domain)

      graph = {"nodes": nodes, "edges": edges}
      _save_graph_cache(graph)
      return graph

  def _build_edges(domain: Optional[str] = None) -> list[dict]:
      conn = get_connection()
      where = ""
      params: list = []
      if domain:
          where = "WHERE domain = ?"
          params = [domain]
      rows = conn.execute(
          f"SELECT concepts FROM knowledge_items {where}", params
      ).fetchall()
      edge_map: dict[tuple[str, str], int] = {}
      for row in rows:
          concepts = json.loads(row[0]) if row[0] else []
          for i, c1 in enumerate(concepts):
              for c2 in concepts[i+1:]:
                  key = tuple(sorted([c1, c2]))
                  edge_map[key] = edge_map.get(key, 0) + 1
      return [{"source": k[0], "target": k[1], "weight": v} for k, v in edge_map.items()]

  def _get_cached_graph(self) -> Optional[dict]:
      conn = get_connection()
      row = conn.execute(
          "SELECT graph_data, updated_at FROM knowledge_graph ORDER BY id DESC LIMIT 1"
      ).fetchone()
      return dict(row) if row else None

  def _cache_fresh(self, cached: dict, ttl_minutes: int) -> bool:
      updated = datetime.fromisoformat(cached["updated_at"].replace("Z", "+00:00"))
      return datetime.now(timezone.utc) - updated < timedelta(minutes=ttl_minutes)

  def _filter_by_domain(self, graph: dict, domain: str) -> dict:
      node_ids = {n["id"] for n in graph["nodes"] if n.get("domain") == domain}
      return {
          "nodes": [n for n in graph["nodes"] if n.get("domain") == domain],
          "edges": [e for e in graph["edges"]
                    if e["source"] in node_ids and e["target"] in node_ids],
      }

  def _save_graph_cache(self, graph: dict) -> None:
      conn = get_connection()
      conn.execute(
          "INSERT INTO knowledge_graph (graph_data, updated_at) VALUES (?, ?)",
          (json.dumps(graph), datetime.now(timezone.utc).isoformat()),
      )
  ```
  注意：`_get_cached_graph` 不应该是 self 方法（模块级函数），移除 self 参数。
