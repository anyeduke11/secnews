```python
import tomllib
from pathlib import Path

_GATES_PATH = Path(__file__).resolve().parent.parent / "config" / "feature_gates.toml"

def _load_gates() -> dict:
    with open(_GATES_PATH, "rb") as f:
        return tomllib.load(f)["extensions"]

def is_extension_enabled(name: str) -> bool:
    return _load_gates().get(name, False)

# 扩展→router 映射（每个 router 在 backend.api 中的模块名）
EXTENSION_ROUTERS = {
    "codegarden": [

    "mcp": [

        "mcp_adapters",       # backend.api.mcp_adapters

    "security_graph": ["security", "kl_metrics", "kl_rollback"],  # 部分安全相关路由

def get_enabled_extensions() -> list[str]:
    gates = _load_gates()

def get_extension_routers(name: str) -> list[str]:
```

### 1.6 复利驱动器①：即时分类（采集 tail 后）

```python
# 在 backend/scheduler/jobs.py 的 collect_all_job 尾部 post-ingest 链新增:

async def collect_all_job() -> None:
    # …… 现有采集逻辑不变 ……
    # ---- post-ingest 链 ----
    await trend_rebuild_job()
    await fts_rebuild_job()
    await security_enrichment_job()
    await url_content_check_job()
    await export_rebuild_job()
    # ---- 新增: 即时分类 (复利驱动器①) ----
    await _classify_new_items()

async def _classify_new_items() -> None:
    """采集 tail 后即时分类新 items（不等待 30min 周期）"""
    try:
        from backend.repository.db import get_connection
        from backend.services.auto_classifier import batch_classify
        from backend.repository.knowledge_repo import knowledge_repo

        conn = get_connection()
        # 只取最近 5 分钟内新入库、且未分类的 items
        rows = conn.execute(
            "SELECT id, title, tags, source_url, domain, type, difficulty "
            "FROM knowledge_items "
            "WHERE (domain IS NULL OR type IS NULL OR difficulty IS NULL) "
            "AND ingested_at > datetime('now', '-5 minutes', 'utc') "
            "ORDER BY ingested_at ASC LIMIT 50"
        ).fetchall()
        if not rows:
            return
        items = [dict(r) for r in rows]
        classified = batch_classify(items)
        # 回写分类结果到 DB
        for item in classified:
            knowledge_repo.update_item(item["id"], {
                "domain": item.get("domain"),
                "topic": item.get("topic"),
                "type": item.get("type"),
                "difficulty": item.get("difficulty"),
            })
        logger.info(f"_classify_new_items: {len(classified)} items classified")
    except Exception as e:
        logger.error(f"_classify_new_items crashed: {e}")
```

**验收**: 采集完成后，新 items 的 `domain` 字段非 null

**受影响文件**:

```python
async def create_attention_event(req: AttentionEventRequest) -> dict:
```

### 2.5 SM-2 推送前端消费端（Header 徽标）

```typescript

// 连接到 /api/events SSE
// 收到 review_due 事件 → 显示 "N 条待复习" 徽标
// 点击徽标 → 跳转到 /knowledge?mode=review

import { useEffect, useState } from 'react';

function ReviewDueBadge() {
  const [dueCount, setDueCount] = useState(0);

  useEffect(() => {
    const evtSource = new EventSource('/api/events');
    evtSource.addEventListener('review_due', (e) => {
      const data = JSON.parse(e.data);
      setDueCount(data.count);
    });
    return () => evtSource.close();
  }, []);

  if (dueCount === 0) return null;
  return (
    <span className="ml-2 inline-flex items-center px-2 py-0.5 rounded-full text-xs font-medium bg-error text-white">
```

## 风险与预案

| 风险 | 概率 | 影响 | 预案 |
|------|------|------|------|
| `feature_gates.toml` 读取失败导致启动崩溃 | 低 | 高 | `_load_gates()` 加 try/except，回退到"全部 extension 开启"（保守降级） |
| `_classify_new_items()` 阻塞采集 tail | 中 | 中 | 用 `asyncio.to_thread` 把同步 SQL 放线程池，限时 5s |
| SM-2 推送 SSE 事件量过大 | 低 | 低 | 只推送 count 而非完整列表（现有设计已限制 20 条） |
| 前端 `useFeatureFlags` 与后端配置不同步 | 中 | 低 | 缓存 5 分钟自动刷新，配置变化后手动刷新页面即可 |
| 前端路由懒加载 + 条件渲染导致首次加载变慢 | 低 | 低 | 路由懒加载已有，条件渲染只是 `&&` 短路，不增加 bundle 大小 |
| 注意力→复习转化同一 item 重复创建 | 中 | 低 | 幂等检查（24h 内同一 item 不重复创建） |
| 地图重建 4153 items 的 SQL 查询太慢 | 低 | 中 | `map_updater.update_map()` 已用 `ORDER BY domain` 分批，如仍慢加 LIMIT 500 |
